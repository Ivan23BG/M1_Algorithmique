% ----- Consignes exo 7 ----- %
\begin{td-exo}[Sur le produit matriciel]\,\\ % 7 
	Appelons \((p,q)\)-matrice une matrice à \(p\) lignes et \(q\) colonnes.
	Nous admettons que le produit d'une \((p,q)\)-matrice par une \((q,r)\)-matrice peut se faire à l'aide de \(pqr\) opérations.

	\begin{enumerate}
		\item Soit \(M_i\) une \((p_i, p_{i+1})\)-matrice pour \(i = 1, \ldots, k\).
		Nous effectuons le produit suivant:
		\begin{equation*}
			\left(
				M1 \left(
					M2 \left(
						\ldots \left(
							M_{k-1} M_k
						\right)
					\right)
				\right)
			\right)
		\end{equation*}
		et
		\begin{equation*}
			\left(
				\left(
					\ldots \left(
						M1 M2
					\right) M3
				\right) \ldots M_k
			\right)
		\end{equation*}
		Evaluer le nombre d'opérations nécessaires pour chacun de ces produits.
		Que remarque-t-on?

		\item Montrer que le nombre \(c(k)\) de parenthésages possibles d'un produit de \(k\) matrices vérifie, si on pose \(c(1) = 1\), 
		\begin{equation*}
			c(k) = \sum_{i=1}^{k-1} c(i) c(k - i)
		\end{equation*}
		Nous admettrons que \(c(k)\) est de l'ordre de \(\frac{4^{k - 1}}{k \sqrt{\pi k}}\) (ceci peut être obtenu en résolvant la récurrence puis à l'aide de la formule de Stirling).
		Une description exhaustive de tous les parenthésages d'un produit de \(k\) matrices, dans le but de déterminer celui qui permet d'effectuer le produit avec un nombre minimum d'opérations, nécessite donc un temps de calcul déraisonnablement exponentiel.
		La programmation dynamique permet de résoudre ce problème (d'un parenthésage optimum) en temps polynomial.

		\item En considérant les blocs \(M_{i, j} = M_i M_{i + 1} \ldots M_j\) pour \(1 \leq i \leq j \leq k\), et en désignant par \(m_{i, j}\) le coût minimum du calcul du produit \(M_{i, j}\), établir la récurrence permettant d'appliquer la programmation dynamique pour obtenir \(m_{1, k}\) et le parenthésage correspondant.

		\item Appliquer ce principe pouur \(k = 5\) et \(p_1 = 1, p_2 = 2, p_3 = 5, p_4 = 10, p_5 = 4\) et \(p_x = 100\) (nous n'omettrons pas de construire la table correspondante et d'indiquer précisément le parenthésage optimal).
	\end{enumerate}
\end{td-exo}

% ----- Solutions exo 7 ----- %
\iftoggle{showsolutions}{ 
	\begin{td-sol}[]\ % 7
		\begin{enumerate}
			\item Le nombre d'opérations nécessaires pour le premier produit est:
			\begin{equation*}
				\begin{aligned}
					&\;\;\;\;\;
					\begin{aligned}
						(k - 1) k (k + 1) &+ (k - 2) (k - 1) (k + 1) \\
						&+ (k - 3) (k - 2) (k + 1) \\
						&+ \ldots \\
						&+ 2 \cdot 3 \cdot (k + 1) \\
						&+ 1 \cdot 2 \cdot (k + 1) 
					\end{aligned}\\
					&= (k + 1) \sum_{i = 1}^{k - 1} i (i + 1) \\
					&= (k + 1) \sum_{i = 1}^{k - 1} (i^2 + i) \\
					&= (k + 1) \left( \frac{(k - 1) k (2k - 1)}{6} + \frac{(k - 1) k}{2} \right) \\
					&= (k + 1) \frac{(k - 1) k (2k + 2)}{6} \\
					&= \frac{(k - 1) k (k + 1)^2}{6}\\
					&= O(k^4)
				\end{aligned}
			\end{equation*}
			Le nombre d'opérations nécessaires pour le second produit est:
			\begin{equation*}
				\begin{aligned}
					&\;\;\;\;\;
					\begin{aligned}
						1 \cdot 2 \cdot 3 &+ 1 \cdot 3 \cdot 4 \\
						&+ \ldots \\
						&+ 1 \cdot (k - 1) \cdot k \\
						&+ 1 \cdot k \cdot (k + 1)
					\end{aligned}\\
					&= \sum_{i = 2}^{k} i (i + 1) \\
					&= \sum_{i = 2}^{k} (i^2 + i) \\
					&= \frac{k (k + 1) (2k + 1)}{6} + \frac{k (k + 1)}{2} - 2 \\
					&= O(k^3)
				\end{aligned}
			\end{equation*}
			On remarque que l'ordre des produits a une influence significative sur le nombre d'opérations nécessaires pour effectuer le produit matriciel.
			Ici on note une différence d'ordre \(O(k^4)\) contre \(O(k^3)\).

			\item On peut remarquer que choisir un parenthésage pour un produit de \(k\) matrices revient à choisir le dernier produit à effectuer (et puis de parenthéser les sous-produits), c'est à dire à choisir un \(i\) tel que \(1 \leq i < k\) et à effectuer le produit de \(M_1 M_2 \ldots M_i\) puis le produit de \(M_{i + 1} M_{i + 2} \ldots M_k\).
			Comme il y a \(c(i)\) façons de parenthéser le produit de \(i\) matrices et \(c(k - i)\) façons de parenthéser le produit de \(k - i\) matrices, il y a \(c(i) c(k - i)\) façons de parenthéser le produit de \(k\) matrices en choisissant \(i\) comme dernier produit à effectuer.
			En sommant sur tous les \(i\) possibles, on obtient la récurrence suivante:
			\begin{equation*}
				c(k) = \sum_{i = 1}^{k - 1} c(i) c(k - i)
			\end{equation*}
			
			\item En choisissant \(i\) comme dernier produit à effectuer pour le produit de \(M_i M_{i + 1} \ldots M_j\), on doit effectuer le produit de \(M_i M_{i + 1} \ldots M_k\) puis le produit de \(M_{k + 1} M_{k + 2} \ldots M_j\) puis le produit de ces deux produits.
			Le coût de ce parenthésage est donc \(m_{i, k} + m_{k + 1, j} + p_i p_{k + 1} p_{j + 1}\).
			En minimisant sur tous les \(i\) possibles, on obtient la récurrence suivante:
			\begin{equation*}
				m_{i, j} = \min_{i \leq k < j} \left( m_{i, k} + m_{k + 1, j} + p_i p_{k + 1} p_{j + 1} \right)
			\end{equation*}
			avec \(m_{i, i} = 0\) pour tout \(i\).

			\item On travaille donc avec des matrices de tailles:
			\begin{equation*}
				\begin{aligned}
					&M_1: 1 \times 2, M_2: 2 \times 5,\\
					&M_3: 5 \times 10, M_4: 10 \times 4,\\
					&M_5: 4 \times 100
				\end{aligned}
			\end{equation*}
			Cas de base: \(m_{i, i} = 0\) pour \(i = 1, \ldots, 5\).
			
			Cas de \(m_{i, j}\) pour \(j = i + 1\):
			\begin{equation*}
				\begin{aligned}
					&m_{1, 2} = 1 \cdot 2 \cdot 5 = 10, \\
					&m_{2, 3} = 2 \cdot 5 \cdot 10 = 100,\\
					&m_{3, 4} = 5 \cdot 10 \cdot 4 = 200, \\
					&m_{4, 5} = 10 \cdot 4 \cdot 100 = 4000
				\end{aligned}
			\end{equation*}
			Cas de \(m_{i, j}\) pour \(j = i + 2\):
			\begin{equation*}
				\begin{aligned}
					&\begin{aligned}
						m_{1, 3} &= \min( m_{1, 1} + m_{2, 3} + 1 \cdot 2 \cdot 10, m_{1, 2} + m_{3, 3} + 1 \cdot 5 \cdot 10 ) \\
						&= \min(120, 60) = 60,
					\end{aligned}\\
					&\begin{aligned}
						m_{2, 4} &= \min( m_{2, 2} + m_{3, 4} + 2 \cdot 5 \cdot 4, m_{2, 3} + m_{4, 4} + 2 \cdot 10 \cdot 4 ) \\
						&= \min(240, 180) = 180,
					\end{aligned}\\
					&\begin{aligned}
						m_{3, 5} &= \min( m_{3, 3} + m_{4, 5} + 5 \cdot 10 \cdot 100, m_{3, 4} + m_{5, 5} + 5 \cdot 4 \cdot 100 ) \\
						&= \min(9000, 2200) = 2200
					\end{aligned}
				\end{aligned}
			\end{equation*}
			Cas de \(m_{i, j}\) pour \(j = i + 3\):
			\begin{equation*}
				\begin{aligned}
					&\begin{aligned}
						m_{1, 4} 
						&= \min( m_{1, 1} + m_{2, 4} + 1 \cdot 2 \cdot 4, m_{1, 2} + m_{3, 4} + 1 \cdot 5 \cdot 4, m_{1, 3} + m_{4, 4} + 1 \cdot 10 \cdot 4 ) \\
						&= \min(188, 230, 100) = 100,
					\end{aligned}\\
					&\begin{aligned}
						m_{2, 5} 
						&= \min( m_{2, 2} + m_{3, 5} + 2 \cdot 5 \cdot 100, m_{2, 3} + m_{4, 5} + 2 \cdot 10 \cdot 100, m_{2, 4} + m_{5, 5} + 2 \cdot 4 \cdot 100 ) \\
						&= \min(3200, 6100, 980) = 980
					\end{aligned}
				\end{aligned}
			\end{equation*}
			Cas de \(m_{i, j}\) pour \(j = i + 4\):
			\begin{equation*}
				\begin{aligned}
					m_{1, 5} 
					&= \min\left(
						\begin{aligned}
							m_{1, 1} + m_{2, 5} + 1 \cdot 2 \cdot 100, \\
							m_{1, 2} + m_{3, 5} + 1 \cdot 5 \cdot 100, \\
							m_{1, 3} + m_{4, 5} + 1 \cdot 10 \cdot 100, \\
							m_{1, 4} + m_{5, 5} + 1 \cdot 4 \cdot 100 
						\end{aligned}\right)\\
					&= \min(1180, 2710, 5060, 500) \\
					&= 500
				\end{aligned}
			\end{equation*}
			La table correspondante est donc:
			\begin{center}
				\begin{tabular}{c|ccccc}
					\(i \backslash j\) & \(1\) & \(2\) & \(3\) & \(4\) & \(5\) \\
					\hline
					\(1\) & \(0\) & \(10\) & \(60\) & \(100\) & \(500\) \\
					\(2\) &  & \(0\) & \(100\) & \(180\) & \(980\) \\
					\(3\) &  &  & \(0\) & \(200\) & \(2200\) \\
					\(4\) &  &  &  & \(0\) & \(4000\) \\
					\(5\) &  &  &  &  & \(0\)
				\end{tabular}
			\end{center}
			Pour obtenir le parenthésage optimal, il suffit de regarder les couts min à chaque étape.
			Pour minimiser le produit de \(M_1 M_2 M_3 M_4 M_5\), on effectue d'abord \((M_1 \cdots M_4) M_5\) (cout min).
			Pour minimiser le produit de \(M_1 M_2 M_3 M_4\), on effectue d'abord \((M_1 M_2 M_3) M_4\) (cout min).
			Pour minimiser le produit de \(M_1 M_2 M_3\), on effectue d'abord \((M_1 M_2) M_3\) (cout min).
			Le parenthésage optimal est donc:
			\begin{equation*}
				\left(
					\left(
						\left(
							M_1 M_2
						\right) M_3
					\right) M_4
				\right) M_5
			\end{equation*}
		\end{enumerate}
	\end{td-sol}
}{}
